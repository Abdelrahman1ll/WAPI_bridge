أمثلة الربط (Code Snippets)

استخدم مكتبة axios لإرسال الرسائل بسهولة:

JavaScript (Node.js) 
const axios = require('axios');

const sendMessage = async () => {
  try {
    const response = await axios.post('API_URL/external/send/USER_ID', {
      phone: '201xxxxxxxxx',
      message: 'Hello from WhatsApp API!'
    }, {
      headers: {
        'Authorization': 'Bearer YOUR_TOKEN'
      }
    });
    console.log(response.data);
  } catch (error) {
    console.error(error);
  }
};

استخدم مكتبة requests لإرسال الرسائل بسهولة:

Python 
import requests

def send_message():
    url = "API_URL/external/send/USER_ID"
    headers = {
        "Authorization": "Bearer YOUR_TOKEN",
        "Content-Type": "application/json"
    }
    payload = {
        "phone": "201xxxxxxxxx",
        "message": "Hello from Python!"
    }
    
    response = requests.post(url, json=payload, headers=headers)
    print(response.json())
